\section{Introducción}
%==============================================================================

%------------------------------------- 3. MOTIVACIÓN, PREGUNTA E HIPÓTESIS --
\begin{frame}{Motivación, pregunta e hipótesis}
  \footnotesize
  \begin{itemize}
    \item La participación laboral femenina determina el bienestar de los
          hogares y el crecimiento en América Latina
          \citep{martinoty_partners_2022}; una barrera central es la
          \alert{falta de cuidado infantil accesible}
          \citep{rodriguez_chamussy_educacion_2017}.
    \item El \textbf{Programa Nacional Cuna Más} (MIDIS) da cuidado diurno a la
          primera infancia y reduce el costo de oportunidad del tiempo de las
          madres \citep{attanasio_community_2013}.
  \end{itemize}

  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{\small Pregunta de investigación}
        \scriptsize ¿Cuál es el impacto de participar en Cuna Más sobre la
        participación laboral de las mujeres de 14 a 50 años en el Perú
        durante 2024?
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{exampleblock}{\small Hipótesis}
        \scriptsize El efecto es \textbf{positivo}: las mujeres cuyo hogar
        accedió al servicio tienen mayor probabilidad de pertenecer a la PEA.
      \end{exampleblock}
    \end{column}
  \end{columns}

  \vspace{0.15cm}
  \begin{alertblock}{\small Problema de identificación}
    \scriptsize El acceso \textbf{no es aleatorio}: depende de la focalización del
    MIDIS y de la decisión de cada familia. Por eso se usan tres estrategias:
    \textbf{MCO} $\rightarrow$ \textbf{PSM} (selección en observables)
    $\rightarrow$ \textbf{VI} (selección en no observables).
  \end{alertblock}
\end{frame}

\section{Propensity Score Matching}
%==============================================================================

%------------------------------------------- 8. METODOLOGÍA Y SOPORTE COMÚN --
\begin{frame}{PSM: metodología y soporte común}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \scriptsize
      El PSM compara beneficiarias con no beneficiarias de \textbf{probabilidad
      estimada de acceder similar}, condicional en observables
      \citep{rosenbaum_central_1983}. El parámetro estimado es el ATT:
      \begin{equation*}
        \text{ATT} = \mathbb{E}\big[\,Y_1 - Y_0 \mid D=1\,\big]
      \end{equation*}
      \vspace{-0.35cm}
      \begin{equation*}
        = \mathbb{E}\Big\{\mathbb{E}\big[Y \mid D=1, p(\mathbf{X})\big] -
        \mathbb{E}\big[Y \mid D=0, p(\mathbf{X})\big] \Big| D=1\Big\}
      \end{equation*}
      \begin{itemize}
        \item $p(\mathbf{X}) = \Pr(D=1 \mid \mathbf{X})$ se estima por un
              \textbf{logit} con covariables que anteceden al programa
              (supuesto CIA).
        \item Se impone \textbf{soporte común}:
              $0 < \Pr(D=1\mid\mathbf{X}) < 1$, descartando tratados fuera del
              rango del grupo de comparación.
        \item Alta \textbf{superposición} de densidades: se cumple el
              \emph{overlapping} y la comparación entre grupos es válida.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{figure}
        \centering
        \includegraphics[width=\linewidth,height=0.62\textheight,keepaspectratio]{graf4_pscore.png}
        \caption{\tiny Soporte común entre tratados y no tratados}
      \end{figure}
    \end{column}
  \end{columns}
\end{frame}

  \scriptsize
  El PSM no resuelve la selección en \textbf{no observables}. Primera etapa:
  \vspace{-0.1cm}
  \begin{equation*}
    \text{cunamas}_i = \pi_0 + \pi_1\,\text{intensidad\_cunamas}_i +
    \pi_2\,\text{lengua\_originaria}_i + \mathbf{X}_i'\boldsymbol{\pi}_3 + v_i
  \end{equation*}

  \vspace{-0.15cm}
  \begin{columns}[T]
    \begin{column}{0.42\textwidth}
      \centering
      \scriptsize
      \setlength{\tabcolsep}{3pt}
      \begin{tabular}{lcc}
        \toprule
        \textbf{Variables} & \textbf{Coef.\ (MC2E)} & \textbf{EE} \\
        \midrule
        \texttt{cunamas}        & \alert{0.974$^{**}$} & (0.457)     \\
        \texttt{urbano}         & $-$0.150$^{***}$     & (0.0114)    \\
        \texttt{ingreso\_hogar} & 9.88e$-$07$^{***}$   & (1.46e$-$07)\\
        \texttt{jefa\_hogar}    & 0.109$^{***}$        & (0.0107)    \\
        \texttt{educ}           & 0.0253$^{***}$       & (0.00279)   \\
        \texttt{edad}           & 0.0135$^{***}$       & (0.000521)  \\
        Constante               & 0.0701$^{***}$       & (0.0257)    \\
        \midrule
        Observaciones & 9,529 & \\
        $R^2$         & 0.019 & \\
        \bottomrule
      \end{tabular}
      \fuente{Errores estándar entre paréntesis. $^{***}p<0.01$;

    \end{column}
    \begin{column}{0.56\textwidth}
      \scriptsize
      \begin{itemize}
        \item \textbf{\texttt{intensidad\_cunamas}:} cobertura territorial del
              programa. La instalación depende de la focalización del MIDIS, de
              modo que a mayor cobertura del dominio, mayor probabilidad de
              acceder.
        \item \textbf{\texttt{lengua\_originaria}:} la focalización territorial
              tiende a priorizar zonas con mayor población indígena y rural.
        \item El coeficiente de \texttt{cunamas} es 0.974, significativo al
              5\,\%: implicaría un aumento de \alert{97.4 p.p.} Las covariables
              exógenas se mantienen cercanas a MCO, es decir, la
              \textbf{imprecisión se concentra en la variable instrumentada}.
      \end{itemize}
      \begin{alertblock}{\small Magnitud implausible}
        \scriptsize Un efecto de 97.4 p.p.\ es económicamente inverosímil y
        obliga a revisar la validez de los instrumentos.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

    \textbf{Estrategia} & \textbf{Efecto estimado} & \textbf{Significancia} \\
    \midrule
    MCO (modelo de probabilidad lineal)    & 7.80 p.p.         & 1\,\% \\
    PSM (cuatro métodos de emparejamiento) & 6.48 -- 8.64 p.p. & 5\,\% \\
    Variables instrumentales (MC2E)        & 97.4 p.p.         & 5\,\% \\
    \bottomrule
  \end{tabular}

  \vspace{0.3cm}
  \begin{columns}[T]
    \begin{column}{0.49\textwidth}
      \begin{block}{\small Lectura conjunta}
        \scriptsize Las tres estrategias coinciden en \textbf{dirección} y
        \textbf{significancia}. MCO y PSM además convergen en la magnitud, lo
        que evidencia robustez frente a la selección en observables. Se toma el
        \textbf{PSM como referencia} para el tamaño del efecto.
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{alertblock}{\small ¿Por qué desconfiar de los 97.4 p.p.?}
        \scriptsize \texttt{lengua\_originaria} podría afectar la participación
        laboral por vías distintas a Cuna Más ---discriminación laboral, menor
        acceso a mercados formales---, violando la restricción de exclusión. El
        test de sobreidentificación no lo descarta: solo evalúa la validez
        conjunta.
      \end{alertblock}
    \end{column}
  \end{columns}

  \vspace{0.1cm}
  \scriptsize
  \textbf{Limitaciones:} diseño transversal (sin efectos fijos de hogar); solo
  243 tratados (2.47\,\%), lo que limita la precisión, sobre todo en 1-NN; y, de
  ser válido el instrumento, el efecto de VI es un \textbf{LATE} no
  generalizable.
\end{frame}

%----------------------------------------------------- 13. CONCLUSIÓN --------
\begin{frame}{Conclusión}
  \begin{block}{Resultado principal}
    La evidencia ---consistente en dirección y significancia bajo MCO, PSM y
    VI--- respalda que Cuna Más no solo cumple su función de cuidado infantil,
    sino que también actúa como un \textbf{mecanismo de activación laboral} para
    las madres beneficiarias.
  \end{block}

  \begin{itemize}
    \item Magnitud estimada del efecto: \alert{entre 6 y 9 puntos porcentuales}
          en la probabilidad de participación laboral.
    \item Implicancia de política: los servicios públicos de cuidado infantil
          pueden ser también instrumentos de inserción laboral femenina.
    \item Agenda futura: datos de panel y mejores instrumentos para identificar
          el efecto sobre no observables.
  \end{itemize}

\end{frame}

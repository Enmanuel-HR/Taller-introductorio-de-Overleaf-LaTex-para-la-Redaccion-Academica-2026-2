\section{Mecanismos Causales e Hipótesis}
% ==============================================================================

\begin{frame}{Mecanismos Causales de la Iconografía Punitiva}
    \begin{alertblock}{Pregunta Central}
        ¿Cómo influye la exposición a imágenes de castigo y sometimiento policial sobre la cooperación de los ciudadanos?
    \end{alertblock}

    \vspace{0.4cm}
    \pause
    \begin{enumerate}
        \item \textbf{Mecanismo de Legitimidad:} Apelación a símbolos patrios y soberanía popular.
        \pause
        \item \textbf{Mecanismo de Auto-Interés (\textit{Self-Interest}):} Percepción de provisión estatal de seguridad.
        \pause
        \item \textbf{Mecanismo de Coerción Simbólica:} Demostración de poderío físico y disuasión punitiva.
    \end{enumerate}
\end{frame}

\begin{frame}{Hipótesis y Modelo Econométrico}
    \begin{exampleblock}{Hipótesis de Trabajo ($H_1$)}
        La exposición a iconografía populista punitiva incrementa el Índice de Cooperación Punitiva Ciudadana (ICPC).
    \end{exampleblock}

    \vspace{0.3cm}
    \pause
    \textbf{Especificación del Modelo:}
    \begin{equation}
        Y_i = \alpha + \beta \, T_i + \sum_{k=1}^K \gamma_k X_{ki} + \varepsilon_i
    \end{equation}

    \begin{itemize}
        \item $Y_i$: Índice de Cooperación Punitiva Ciudadana (ICPC).
        \item $T_i \in \{0, 1\}$: Asignación aleatoria (Control vs. Tratamiento).
        \item $\beta$: Efecto promedio del tratamiento (\textit{ATE}).
    \end{itemize}
\end{frame}
